WITH m, (m.released - p.born) AS age

LIMIT 10

#18 Quantos atores atuaram num só filme

WITH p, count(m) AS numMovies
WHERE numMovies = 1
RETURN p.name AS Actor

#19 Quantos filmes cujo elenco seja composto exclusivamente por atores 
#   que nunca trabalharam com o mesmo realizador em nenhum outro filme?
MATCH (m:Movie)<-[:ACTED_IN]-(a:Person)
MATCH (m)<-[:DIRECTED]-(d:Person)
WITH m, d, collect(DISTINCT a) AS cast
WHERE none(a IN cast WHERE 
    EXISTS {
        MATCH (a)-[:ACTED_IN]->(:Movie)<-[:DIRECTED]-(d)
        WHERE a <> d
    }
)
RETURN m.title AS Movie
